%% ---------------- Problem statement ----------------
\section[Problem]{Problem Statement}
\begin{frame}{Problem statement}
\textbf{Formally:} for a document $D=\{S_1,\ldots,S_n\}$, predict for every adjacent pair $(S_i,S_{i+1})$:
\[
y_i =
\begin{cases}
0 & S_i \text{ and } S_{i+1} \text{ share an author,}\\
1 & \text{an author switch occurs between them.}
\end{cases}
\]
\end{frame}

\begin{frame}{Why is it interesting?}
\begin{itemize}\itemsep0.7em
\item \textbf{A difficult problem:} author style is subtle, especially when the topic is controlled
\item \textbf{Short evidence:} one sentence carries only a small trace of the author
\item \textbf{Descriptive cues:} short style features (length, punctuation, vocabulary) can still describe an author
\item \textbf{Interpretable:} we can see which cues drive a decision
\item \textbf{Useful:} plagiarism detection and analysis of co-authored text
\end{itemize}
\end{frame}

\begin{frame}{Hashed character trigrams and the pair vector}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Hashing trick}
\begin{itemize}\itemsep0.3em
\item Each trigram $\rightarrow$ one of $B=256$ buckets
\item Advantage: fixed length, no corpus required
\item Cost: there are collisions, and the buckets mean nothing to us
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Pair feature vector: compressed version (11 values)}
\begin{itemize}\itemsep0.3em
\item 9 absolute differences $|z_i^{(k)}-z_{i+1}^{(k)}|$
\item Trigram cosine (kept readable)
\item Overall distance $d_i$
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Models overview}

\begin{tabular}{p{3cm}p{9cm}}
\toprule
\textbf{Model} & \textbf{Setup}\\
\midrule
Decision tree & Gini, depth 5, min 20 to split, 5 per leaf, class weighted, threshold 0.5\\[0.6em]
Logistic regression & Batch gradient descent, 10,000 iterations, lr 0.6, class weighted\\[0.6em]
Neural network & 11--128--64--32--16--8--1, ReLU, mini batch SGD (lr 0.001, batch 64), L2 0.01, dropout 0.2, early stopping\\
\bottomrule
\end{tabular}

\begin{frame}{Interpretation: what the models actually used}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Decision tree}
\begin{itemize}\itemsep0.3em
\item Root: character length difference (idx 1)
\item Then average word length difference (idx 6)
\item Trigram cosine appears at nearly every level
\item Only 5 of 11 features used at all
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Logistic regression weights}
\begin{itemize}\itemsep0.3em
\item Trigram cosine: $-4.71$ (more similar $\Rightarrow$ fewer switches)
\item Character length difference: $+0.99$
\item Overall distance: $+0.72$
\item Frequent word and punctuation differences $\approx 0$
\end{itemize}
\end{column}
\end{columns}
\bigskip
\begin{block}{}
The nine scalars are mostly surface statistics.
\end{block}
\end{frame}
